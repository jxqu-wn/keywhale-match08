-- 行政区划字典表
CREATE TABLE IF NOT EXISTS administrative_division (
    id CHAR(6) PRIMARY KEY COMMENT '行政区划代码',
    name VARCHAR(50) NOT NULL COMMENT '行政区划名称',
    full_name VARCHAR(100) NOT NULL COMMENT '行政区划全称',
    province_name VARCHAR(20) NOT NULL COMMENT '省级名称',
    city_name VARCHAR(20) NOT NULL COMMENT '市级名称',
    county_name VARCHAR(20) NOT NULL COMMENT '县级名称',

    longitude DECIMAL(10, 6) COMMENT '经度',
    latitude DECIMAL(10, 6) COMMENT '纬度'
) ENGINE=InnoDB COMMENT='统一行政区划表';

    level_code VARCHAR(20) PRIMARY KEY COMMENT '学历代码(UNDERGRAD/MASTER等)',
    level_name VARCHAR(50) NOT NULL COMMENT '学历名称'
);

-- 3. 学生状态字典
DROP TABLE IF EXISTS dict_student_status;
CREATE TABLE dict_student_status (
    status_code VARCHAR(20) PRIMARY KEY COMMENT '状态代码(ACTIVE/INACTIVE/GRADUATED等)',
    status_name VARCHAR(50) NOT NULL COMMENT '状态名称'
);

-- 4. 课程类型字典
DROP TABLE IF EXISTS dict_course_type;
CREATE TABLE dict_course_type (
    type_code VARCHAR(20) PRIMARY KEY COMMENT '课程类型代码',
    type_name VARCHAR(50) NOT NULL COMMENT '课程类型名称'
);

-- 5. 考试项目字典
DROP TABLE IF EXISTS dict_exam_item;
CREATE TABLE dict_exam_item (
    item_code VARCHAR(20) PRIMARY KEY COMMENT '考试项目代码',
    item_name VARCHAR(50) NOT NULL COMMENT '考试项目名称',
    default_weight DECIMAL(5,2) COMMENT '默认权重（如平时成绩占30%）'
);

-- 6. 选课方式字典
DROP TABLE IF EXISTS dict_take_mode;
CREATE TABLE dict_take_mode (
    mode_code VARCHAR(20) PRIMARY KEY COMMENT '选课方式代码',
    mode_name VARCHAR(50) NOT NULL COMMENT '选课方式名称'
);

-- 7. 学期字典（与业务库中的 semester 表互补，用于描述学期性质）
DROP TABLE IF EXISTS dict_semester_type;
CREATE TABLE dict_semester_type (
    semester_code VARCHAR(20) PRIMARY KEY COMMENT '学期代码（如2024-FALL）',
    semester_name VARCHAR(50) NOT NULL COMMENT '学期名称',
    season VARCHAR(10) COMMENT '季节(SPRING/FALL/SUMMER)',
    is_current TINYINT(1) COMMENT '是否为当前学期'
);

-- 8. 成绩等级参考表
DROP TABLE IF EXISTS dict_grade_level;
CREATE TABLE dict_grade_level (
    grade_level CHAR(2) PRIMARY KEY COMMENT '等级(A/B/C/D/F)',
    min_score INT COMMENT '最低分数',
    max_score INT COMMENT '最高分数',
    grade_point DECIMAL(3,2) COMMENT '绩点'
);
