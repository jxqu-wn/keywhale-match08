-- =====================================================
-- 交通库相关表
CREATE DATABASE IF NOT EXISTS transportation;
USE transportation;
-- 1. 车辆公司表
CREATE TABLE vehicle_company (
    company_id VARCHAR(20) PRIMARY KEY COMMENT '车辆公司编号',
    company_name VARCHAR(100) NOT NULL COMMENT '公司名称',
    business_type VARCHAR(50) NOT NULL COMMENT '经营类型（客运/货运/危化品运输）',
    establish_date DATE COMMENT '成立日期',
    legal_person VARCHAR(50) COMMENT '法人',
    contact_phone VARCHAR(20) COMMENT '联系电话',
    address VARCHAR(200) COMMENT '公司地址',
    area_id CHAR(6) NOT NULL COMMENT '注册地区划代码',
    status ENUM('运营','停业','注销') DEFAULT '运营' COMMENT '经营状态',
    FOREIGN KEY (area_id) REFERENCES common.administrative_division(id)
) ENGINE=InnoDB COMMENT='车辆运输公司信息表';

-- 2. 驾驶员表（车辆公司的员工）
CREATE TABLE driver (
    driver_id VARCHAR(18) PRIMARY KEY COMMENT '身份证号',
    name VARCHAR(50) NOT NULL COMMENT '姓名',
    gender ENUM('男','女') NOT NULL COMMENT '性别',
    birth_date DATE NOT NULL COMMENT '出生日期',
    license_type VARCHAR(5) NOT NULL COMMENT '驾照类型代码',
    issue_date DATE NOT NULL COMMENT '发证日期',
    valid_years TINYINT NOT NULL COMMENT '有效期(年)',
    phone VARCHAR(20) COMMENT '联系电话',
    address VARCHAR(200) COMMENT '联系地址',
    area_id CHAR(6) NOT NULL COMMENT '行政区划代码',
    status ENUM('正常','暂扣','吊销','过期') DEFAULT '正常' COMMENT '驾照状态',
    company_id VARCHAR(20) NOT NULL COMMENT '所属车辆公司ID',
    INDEX idx_name (name),
    INDEX idx_license (license_type),
    FOREIGN KEY (area_id) REFERENCES common.administrative_division(id),
    FOREIGN KEY (license_type) REFERENCES common.license_type_dict(license_code),
    FOREIGN KEY (company_id) REFERENCES vehicle_company(company_id)
) ENGINE=InnoDB COMMENT='驾驶员信息表（车辆公司员工）';

-- 3. 车辆登记表
CREATE TABLE vehicle (
    vehicle_id VARCHAR(20) PRIMARY KEY COMMENT '车辆唯一编号',
    plate_number VARCHAR(10) NOT NULL UNIQUE COMMENT '车牌号',
    owner_id VARCHAR(18) NOT NULL COMMENT '车主（驾驶员）身份证号',
    vehicle_type_code VARCHAR(20) NOT NULL COMMENT '车辆类型代码',
    brand VARCHAR(30) NOT NULL COMMENT '品牌',
    model VARCHAR(30) NOT NULL COMMENT '型号',
    manufacture_year YEAR NOT NULL COMMENT '生产年份',
    load_capacity DECIMAL(8,2) COMMENT '载重(吨)',
    seat_count TINYINT UNSIGNED COMMENT '座位数（客车用）',
    color VARCHAR(20) COMMENT '颜色',
    register_date DATE NOT NULL COMMENT '注册日期',
    area_id CHAR(6) NOT NULL COMMENT '注册地区划代码',
    insurance_expire DATE COMMENT '保险到期日',
    company_id VARCHAR(20) NOT NULL COMMENT '所属车辆公司ID',
    INDEX idx_plate (plate_number),
    INDEX idx_owner (owner_id),
    INDEX idx_type (vehicle_type_code),
    FOREIGN KEY (owner_id) REFERENCES driver(driver_id),
    FOREIGN KEY (area_id) REFERENCES common.administrative_division(id),
    FOREIGN KEY (vehicle_type_code) REFERENCES common.vehicle_type_dict(type_code),
    FOREIGN KEY (company_id) REFERENCES vehicle_company(company_id)
) ENGINE=InnoDB COMMENT='车辆登记表';

-- 4. 船舶公司表（独立）
CREATE TABLE shipping_company (
    company_id VARCHAR(20) PRIMARY KEY COMMENT '船舶公司编号',
    company_name VARCHAR(100) NOT NULL COMMENT '公司名称',
    business_type VARCHAR(50) NOT NULL COMMENT '经营类型（集装箱/散货/油轮/客运）',
    establish_date DATE NOT NULL COMMENT '成立日期',
    legal_person VARCHAR(50) NOT NULL COMMENT '法人',
    contact_phone VARCHAR(20) COMMENT '联系电话',
    address VARCHAR(200) COMMENT '公司地址',
    area_id CHAR(6) NOT NULL COMMENT '行政区划代码',
    status ENUM('运营','停业','注销') DEFAULT '运营' COMMENT '经营状态',
    FOREIGN KEY (area_id) REFERENCES common.administrative_division(id)
) ENGINE=InnoDB COMMENT='船舶公司信息表';

-- 5. 员工表（船舶公司的员工）
CREATE TABLE employee (
    employee_id VARCHAR(18) PRIMARY KEY COMMENT '身份证号',
    name VARCHAR(50) NOT NULL COMMENT '姓名',
    gender ENUM('男','女') NOT NULL COMMENT '性别',
    employee_type_code VARCHAR(20) NOT NULL COMMENT '人员类型代码（船长/大副等）',
    company_id VARCHAR(20) NOT NULL COMMENT '所属船舶公司ID',
    hire_date DATE COMMENT '入职日期',
    phone VARCHAR(20) COMMENT '联系电话',
    status ENUM('在职','离职','退休') DEFAULT '在职' COMMENT '状态',
    FOREIGN KEY (employee_type_code) REFERENCES common.employee_type_dict(type_code),
    FOREIGN KEY (company_id) REFERENCES shipping_company(company_id)
) ENGINE=InnoDB COMMENT='船舶公司员工表';

-- 6. 船舶基础信息表（船舶公司的子表）（通过mmsi关联Postgre中的ais数据）
CREATE TABLE vessel (
    mmsi INT UNSIGNED NOT NULL COMMENT '船舶MMSI，9位唯一标识',
    company_id VARCHAR(20) NOT NULL COMMENT '所属船舶公司ID',
    vessel_type SMALLINT COMMENT '船舶类型代码（参照AIS标准）',
    max_load DECIMAL(10,2) COMMENT '最大荷载量（吨）',
    max_speed DECIMAL(4,1) COMMENT '最大航速（节）',
    endurance VARCHAR(32) COMMENT '自持力（天/海里）',
    captain_name VARCHAR(32) COMMENT '船长姓名',
    area_id CHAR(6) NOT NULL COMMENT '注册地区划代码',
    UNIQUE KEY uk_mmsi (mmsi),
    FOREIGN KEY (company_id) REFERENCES shipping_company(company_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (area_id) REFERENCES common.administrative_division(id),
    FOREIGN KEY (vessel_type) REFERENCES common.vessel_type_dict(type_code)
) ENGINE=InnoDB COMMENT='船舶基础信息表';

CREATE DATABASE IF NOT EXISTS finance_tax;
USE finance_tax;

-- 公司主表
CREATE TABLE companies (
    company_id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '公司唯一ID',
    company_name VARCHAR(200) NOT NULL UNIQUE COMMENT '公司法定全称',
    stock_code VARCHAR(20) UNIQUE COMMENT '股票代码（若上市）',
    established_year YEAR COMMENT '注册/成立年份',
    registered_capital DECIMAL(18,2) COMMENT '注册资本（万元）',
    industry_code VARCHAR(20) COMMENT '所属行业代码',
    registration_type_code VARCHAR(20) COMMENT '登记注册类型代码',
    is_private BOOLEAN COMMENT '是否民营企业',
    size_category_code VARCHAR(10) COMMENT '规模划型代码',
    is_key_high_tech BOOLEAN DEFAULT FALSE COMMENT '是否重点扶持高新技术企业',
    is_tech_advanced_service BOOLEAN DEFAULT FALSE COMMENT '是否技术先进型服务企业',

    FOREIGN KEY (industry_code) REFERENCES common.dict_industry(industry_code),
    FOREIGN KEY (registration_type_code) REFERENCES common.dict_registration_type(type_code),
    FOREIGN KEY (size_category_code) REFERENCES common.dict_size_category(size_code)
) COMMENT='公司基础信息主表';

-- 股票信息表
CREATE TABLE stocks (
    stock_id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '词条ID',
    company_id BIGINT NOT NULL COMMENT '关联公司ID',
    stock_code VARCHAR(20) NOT NULL UNIQUE COMMENT '股票代码',
    stock_name VARCHAR(50) NOT NULL COMMENT '股票简称',
    stock_full_name VARCHAR(200) COMMENT '公司全称（冗余）',
    share_right_type VARCHAR(20) DEFAULT 'COMMON' COMMENT '股东权利类型（引用字典）',
    register_type VARCHAR(20) DEFAULT 'REGISTERED' COMMENT '记名方式',
    par_value_type VARCHAR(20) DEFAULT 'PAR' COMMENT '面额类型',
    market_class VARCHAR(5) COMMENT '上市地点分类（A/B/H/N等）',
    listing_date DATE COMMENT '上市时间',
    total_shares BIGINT NOT NULL COMMENT '股本（股）',
    current_price DECIMAL(10,2) COMMENT '最新股票价格（元）',

    FOREIGN KEY (company_id) REFERENCES companies(company_id)
) COMMENT='股票基本信息表';

-- 股票日成交信息表
CREATE TABLE stock_daily_trades (
    trade_id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '词条ID',
    stock_id BIGINT NOT NULL COMMENT '关联股票ID',
    trade_date DATE NOT NULL COMMENT '交易日期',
    volume BIGINT NOT NULL COMMENT '成交量（股）',
    avg_trade_size INT COMMENT '每笔成交股数',
    open_price DECIMAL(10,2) NOT NULL COMMENT '开盘价',
    close_price DECIMAL(10,2) NOT NULL COMMENT '收盘价',
    high_price DECIMAL(10,2) NOT NULL COMMENT '最高价',
    low_price DECIMAL(10,2) NOT NULL COMMENT '最低价',

    FOREIGN KEY (stock_id) REFERENCES stocks(stock_id),
    UNIQUE KEY uk_stock_date (stock_id, trade_date)
) COMMENT='股票每日成交汇总信息';

-- 公司财务信息表
CREATE TABLE company_financials (
    financial_id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '词条ID',
    company_id BIGINT NOT NULL COMMENT '关联公司ID',
    report_date DATE NOT NULL COMMENT '报表日期',
    dividend_cash DECIMAL(18,2) DEFAULT 0 COMMENT '现金分红总额（万元）',
    dividend_stock INT DEFAULT 0 COMMENT '每股股息（分/股）',
    total_assets DECIMAL(18,2) NOT NULL COMMENT '总资产（万元）',
    total_liabilities DECIMAL(18,2) NOT NULL COMMENT '总负债（万元）',

    FOREIGN KEY (company_id) REFERENCES companies(company_id),
    UNIQUE KEY uk_company_report (company_id, report_date)
) COMMENT='公司定期财务摘要信息';

-- 企业所得税申报信息表
CREATE TABLE corporate_income_tax_returns (
    return_id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '申报ID',
    company_id BIGINT NOT NULL COMMENT '关联公司ID',
    tax_year YEAR NOT NULL COMMENT '纳税年度',
    revenue DECIMAL(18,2) NOT NULL COMMENT '收入总额（万元）',
    non_taxable_income DECIMAL(18,2) DEFAULT 0 COMMENT '不征税收入与免税收入（万元）',
    deductions DECIMAL(18,2) NOT NULL COMMENT '扣除金额（万元）',
    prior_year_losses DECIMAL(18,2) DEFAULT 0 COMMENT '允许弥补的以前年度亏损（万元）',
    building_asset_value DECIMAL(18,2) COMMENT '房屋建筑物资产原值（万元）',
    employee_count INT COMMENT '从业人数（全年季度平均值）',
    total_assets DECIMAL(18,2) COMMENT '资产总额（万元，全年季度平均值）',
    is_restricted_industry BOOLEAN DEFAULT FALSE COMMENT '是否从事国家限制或禁止行业',

    FOREIGN KEY (company_id) REFERENCES companies(company_id),
    UNIQUE KEY uk_company_year (company_id, tax_year)
) COMMENT='企业所得税年度申报信息表';

    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '记录ID',
    company_id VARCHAR(20) NOT NULL COMMENT '企业编号',
    cert_type VARCHAR(50) NOT NULL COMMENT '资质类型',

    issue_authority VARCHAR(100) COMMENT '发证机关',
    FOREIGN KEY (company_id) REFERENCES energy_company(company_id)

-- 能源场站表
CREATE TABLE energy_station (
    station_id VARCHAR(20) PRIMARY KEY COMMENT '场站编号',
    station_name VARCHAR(100) NOT NULL COMMENT '场站名称',
    company_id VARCHAR(20) NOT NULL COMMENT '所属企业编号',
    station_type_code VARCHAR(10) NOT NULL COMMENT '场站类型代码',
    energy_type_code VARCHAR(10) NOT NULL COMMENT '能源类型代码',
    installed_capacity DECIMAL(12,2) COMMENT '装机容量(MW)',
    annual_generation DECIMAL(12,2) COMMENT '年发电量(万kWh)',
    grid_connect_date DATE COMMENT '并网日期',
    area_id CHAR(6) NOT NULL COMMENT '所在地区划代码',
    status ENUM('运行','在建','停机','退役') DEFAULT '运行' COMMENT '运行状态',
    FOREIGN KEY (company_id) REFERENCES energy_company(company_id),
    FOREIGN KEY (station_type_code) REFERENCES common.station_type_dict(type_code),
    FOREIGN KEY (energy_type_code) REFERENCES common.energy_type_dict(type_code),
    FOREIGN KEY (area_id) REFERENCES common.administrative_division(id)
) ENGINE=InnoDB COMMENT='能源场站表';

-- 供电区域表

    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '记录ID',

    UNIQUE KEY uk_company_area (company_id, area_id)
) ENGINE=InnoDB COMMENT='供电区域表';

    gender ENUM('男','女') NOT NULL COMMENT '性别',
    birth_date DATE NOT NULL COMMENT '出生日期',

CREATE DATABASE IF NOT EXISTS ecommerce
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_unicode_ci;

USE ecommerce;

CREATE TABLE customer (
  customer_id BIGINT NOT NULL COMMENT '用户ID',
  customer_name VARCHAR(100) NOT NULL COMMENT '用户姓名',
  gender ENUM('M', 'F', 'U') NOT NULL DEFAULT 'U' COMMENT '性别：M男，F女，U未知',
  phone VARCHAR(32) DEFAULT NULL COMMENT '手机号',
  email VARCHAR(128) DEFAULT NULL COMMENT '邮箱',
  level_code VARCHAR(32) NOT NULL COMMENT '用户等级编码，关联common.customer_level_dict.level_code',
  status_code VARCHAR(32) NOT NULL COMMENT '用户状态编码，关联common.customer_status_dict.status_code',
  province_code VARCHAR(12) NOT NULL COMMENT '省级行政区划编码，关联common.administrative_division.division_code',
